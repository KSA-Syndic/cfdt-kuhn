Dans une rubrique chronologique, la gauche va vers le passé (« Plus ancien ») et la droite vers le présent (« Plus récent »), avec la date de chaque article. */ -}} {{- $page := . -}} {{- with partial "navigation/rubrique.html" . }}{{ if $page.IsPage }} {{- $rubrique := . -}} {{- $chrono := .Params.chronologique -}} {{- $pages := partial "navigation/pages-rubrique.html" . -}} {{- $i := -1 }}{{ range $n, $p := $pages }}{{ if eq $p $page }}{{ $i = $n }}{{ end }}{{ end -}} {{- if ge $i 0 }} {{- $avant := false }}{{ if gt $i 0 }}{{ $avant = index $pages (sub $i 1) }}{{ end -}} {{- $apres := false }}{{ if lt (add $i 1) (len $pages) }}{{ $apres = index $pages (add $i 1) }}{{ end -}}
{{- with $avant }} ← {{ if $chrono }}Plus ancien{{ else }}Précédent{{ end }} {{ partial "docs/title" . }} {{- if $chrono }}{{ partial "navigation/date.html" . }}{{ end }} {{- end }} {{- with $apres }} {{ if $chrono }}Plus récent{{ else }}Suivant{{ end }} → {{ partial "docs/title" . }} {{- if $chrono }}{{ partial "navigation/date.html" . }}{{ end }} {{- end }}
{{- end }} {{- end }}{{ end -}}
